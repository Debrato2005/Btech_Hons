import cv2,ultralytics,yaml,numpy as np,json,hashlib,tempfile,shutil

 for i in range(3):w.write(np.full((480,640,3),i*60,dtype=np.uint8))
 w.release();c=cv2.VideoCapture(str(fn));count=0;dims=[]
