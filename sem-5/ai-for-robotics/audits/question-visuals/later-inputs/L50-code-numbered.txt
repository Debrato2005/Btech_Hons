  5 class RobotEmotionEngine:
  6     """
  7     Maintains Albert's simulated internal mood on paired emotion axes.
  8 
  9     Important:
 10     These values simulate an artificial personality. They are not evidence
 11     that the robot genuinely experiences emotions.
 12     """
 13 
 14     def __init__(self):
 15         # Initialize various emotion levels for the robot
 16         self.emotions = {
 17             "happy": 50,
 18             "sad": 50,
 19             "welcome": 50,
 20             "distant": 50,
 21             "friend": 50,
 22             "strange": 50,
 23             "curious": 50,
 24             "frustrated": 50,
 25             "fresh": 80,
 26             "tired": 20,
 27         }
 28 
 29         # Define pairs of opposite emotions
 30         self.opposites = {
 31             "happy": "sad",
 32             "sad": "happy",
 33             "welcome": "distant",
 34             "distant": "welcome",
 35             "friend": "strange",
 36             "strange": "friend",
 37             "curious": "frustrated",
 38             "frustrated": "curious",
 39             "fresh": "tired",
 40             "tired": "fresh",
 41         }
 42 
 43         # Map emotions to angular positions on an emotion axis (in degrees)
 44         self.emotion_axis = {
 45             "happy": 112,
 46             "welcome": 22,
 47             "friend": 67,
 48             "curious": 157,
 49             "sad": 292,
 50             "distant": 202,
 51             "strange": 247,
 52             "frustrated": 337,
 53         }
 54 
 55         # Initialize polar state variables (radius and angle)
 56         self.state_r = 0.0
 57         self.state_theta = 112.0 # Default angle
 58         self.active_mood = "NEUTRAL" # Current active mood
 59         self.update() # Initial update of the emotional state
 60 
 61     def change(self, emotion: str, value: float) -> None:
 62         # Change the value of a specific emotion
 63         if emotion not in self.emotions:
 64             raise KeyError(f"Unknown robot emotion: {emotion}")
 65 
 66         value = float(np.clip(value, 0, 100)) # Clip value to be between 0 and 100
 67         self.emotions[emotion] = value # Set the emotion value
 68 
 69         opposite = self.opposites.get(emotion) # Get the opposite emotion
 70         if opposite is not None:
 71             self.emotions[opposite] = 100.0 - value # Update the opposite emotion value
 72 
 73         self.update() # Recalculate the overall mood
 74 
 75     def nudge(self, emotion: str, amount: float) -> None:
 76         # Adjust an emotion by a small amount
 77         self.change(emotion, self.emotions[emotion] + amount)
 78 
 79     def update(self) -> None:
 80         # Update the robot's overall emotional state (polar coordinates and active mood)
 81         x_sum = 0.0
 82         y_sum = 0.0
 83 
 84         # Calculate sum of x and y components for each emotion vector
 85         for emotion, angle in self.emotion_axis.items():
 86             radius = self.emotions[emotion] - 50.0 # Normalize emotion value around 50
 87             radians = math.radians(angle) # Convert angle to radians
 88             x_sum += radius * math.cos(radians)
 89             y_sum += radius * math.sin(radians)
 90 
 91         self.state_r = math.hypot(x_sum, y_sum) # Calculate resultant radius (magnitude)
 92 
 93         if self.state_r > 1e-3: # If there's a significant emotional state
 94             theta = math.degrees(math.atan2(y_sum, x_sum)) # Calculate resultant angle
 95             self.state_theta = theta % 360 # Normalize angle to 0-360 degrees
 96         else:
 97             self.state_theta = 112.0 # Default angle if no strong emotion
 98 
 99         if self.state_r < 10.0: # If the emotional state is weak, set to neutral
100             self.active_mood = "NEUTRAL"
101             return
102 
103         def angular_distance(a: float, b: float) -> float:
104             # Helper function to calculate shortest angular distance
105             delta = abs(a - b)
106             return min(delta, 360.0 - delta)
107 
108         # Determine the active mood based on the closest emotion axis
109         self.active_mood = min(
110             self.emotion_axis,
111             key=lambda emotion: angular_distance(
112                 self.state_theta,
113                 self.emotion_axis[emotion],
114             ),
115         ).upper()

  5 @dataclass
  6 class ConversationMemory:
  7     """Manages the conversation history and known facts about the user."""
  8 
  9     # Stores known facts about the user, initialized with None values
 10     known_facts: dict = field(
 11         default_factory=lambda: {
 12             "name": None,
 13             "age": None,
 14             "favorite_food": None,
 15             "favorite_book": None,
 16             "favorite_subject": None,
 17             "favorite_song": None,
 18             "favorite_color": None,
 19         }
 20     )
 21     # List of past conversation turns, each a dictionary with 'role' and 'content'
 22     history: list[dict[str, str]] = field(default_factory=list)
 23     # Counter for the number of conversation turns
 24     turn_count: int = 0
 25 
 26     def add_turn(self, user_text: str, assistant_text: str) -> None:
 27         """Adds a user and assistant turn to the conversation history."""
 28         self.history.extend(
 29             [
 30                 {"role": "user", "content": user_text},
 31                 {"role": "assistant", "content": assistant_text},
 32             ]
 33         )
 34 
 35         # Keep the prompt compact for a small local model.
 36         # Only retain the last 12 turns to manage memory usage and context window
 37         self.history = self.history[-12:]
 38         self.turn_count += 1

  5 def analyse_emotion(
  6     user_text: str,
  7     memory: ConversationMemory,
  8 ) -> EmotionAnalysis:
  9     """
 10     Uses the local LLM as a zero-shot, text-only emotion classifier.
 11     """
 12 
 13     # Get the last 6 turns of conversation history for context
 14     recent_context = memory.history[-6:]
 15 
 16     # Construct the prompt for the LLM to classify emotion
 17     prompt = f"""
 18 Classify the emotion expressed in the latest user message.
 19 
 20 Choose exactly one label:
 21 happy, sad, angry, fearful, frustrated, neutral.
 22 
 23 Return:
 24 - emotion
 25 - confidence from 0.0 to 1.0
 26 - valence from -1.0 (very negative) to +1.0 (very positive)
 27 - arousal from -1.0 (very calm) to +1.0 (very activated)
 28 - one short explanation
 29 
 30 Rules:
 31 - Infer only what is supported by the text and recent context.
 32 - Do not diagnose the user.
 33 - If emotional evidence is weak or ambiguous, return neutral with low confidence.
 34 - Treat sarcasm cautiously.
 35 
 36 Recent context:
 37 {json.dumps(recent_context, ensure_ascii=False)}
 38 
 39 Latest message:
 40 {user_text}
 41 """
 42 
 43     try:
 44         # Call the Ollama client to chat with the model
 45         response = client.chat(
 46             model=MODEL_NAME,
 47             messages=[
 48                 {
 49                     "role": "system",
 50                     "content": (
 51                         "You are a conservative text-emotion classifier. "
 52                         "Always follow the supplied JSON schema."
 53                     ),
 54                 },
 55                 {"role": "user", "content": prompt},
 56             ],
 57             format=EmotionAnalysis.model_json_schema(), # Ensure output adheres to EmotionAnalysis schema
 58             options={
 59                 "temperature": 0.0, # Low temperature for consistent classification
 60                 "num_predict": 180, # Max tokens to predict
 61             },
 62             keep_alive="30m", # Keep the model loaded for 30 minutes
 63         )
 64 
 65         # Validate and return the structured emotion analysis
 66         return EmotionAnalysis.model_validate_json(response.message.content)
 67 
 68     except Exception as error:
 69         # Fallback to neutral emotion if LLM call fails
 70         print(f"Emotion-analysis fallback used: {error}")
 71         return EmotionAnalysis(
 72             emotion="neutral",
 73             confidence=0.0,
 74             valence=0.0,
 75             arousal=0.0,
 76             explanation="The local classifier did not return a valid result.",
 77         )

 80 def generate_reply(
 81     user_text: str,
 82     analysis: EmotionAnalysis,
 83     robot: RobotEmotionEngine,
 84     memory: ConversationMemory,
 85 ) -> str:
 86     """Generates a reply from Albert, the robot, based on user input and emotional state."""
 87 
 88     # Filter out unknown facts (None values)
 89     known_facts = {
 90         key: value
 91         for key, value in memory.known_facts.items()
 92         if value is not None
 93     }
 94 
 95     # Construct the system prompt for Albert's persona and context
 96     system_prompt = f"""
 97 You are Albert, a friendly demonstration robot in a beginner AI-for-Robotics course.
 98 
 99 The user's text was estimated as:
100 - emotion: {analysis.emotion}
101 - model-reported confidence: {analysis.confidence:.2f}
102 - valence: {analysis.valence:.2f}
103 - arousal: {analysis.arousal:.2f}
104 
105 Albert's simulated internal mood is: {robot.active_mood}
106 
107 Known user facts:
108 {json.dumps(known_facts, ensure_ascii=False)}
109 
110 Response rules:
111 - Reply naturally in at most three short sentences.
112 - Adapt the tone cautiously to the estimated emotion.
113 - Do not assert that you know exactly how the user feels.
114 - Do not repeatedly name the emotion label.
115 - Ask at most one relevant follow-up question.
116 - Do not claim that Albert truly experiences emotions.
117 - For dangerous or medical situations, recommend contacting a qualified
118   person or emergency service instead of pretending to be a professional.
119 """
120 
121     # Prepare messages for the LLM, including system prompt and recent history
122     messages = [{"role": "system", "content": system_prompt}]
123     messages.extend(memory.history[-8:]) # Include last 8 turns of conversation
124     messages.append({"role": "user", "content": user_text})
125 
126     try:
127         # Call the Ollama client to generate a reply
128         response = client.chat(
129             model=MODEL_NAME,
130             messages=messages,
131             options={
132                 "temperature": 0.55, # Moderate temperature for creative but controlled replies
133                 "num_predict": 140, # Max tokens for the reply
134             },
135             keep_alive="30m", # Keep the model loaded for 30 minutes
136         )
137         return response.message.content.strip()
138 
139     except Exception as error:
140         # Fallback message if LLM call fails
141         return f"I could not reach my local language model. ({error})"

  5 def apply_analysis_to_human_engine(
  6     analysis: EmotionAnalysis,
  7     human: HumanEmotionEngine,
  8 ) -> None:
  9     """
 10     Maps the categorical LLM output and continuous valence/arousal values
 11     onto the human polar state model.
 12     """
 13 
 14     # Use continuous values as the primary mapping to happiness and calmness.
 15     happiness = 50.0 + 50.0 * analysis.valence
 16     calmness = 50.0 - 50.0 * analysis.arousal
 17 
 18     # For low-confidence estimates, keep the emotion closer to neutral.
 19     weight = analysis.confidence
 20     happiness = 50.0 + weight * (happiness - 50.0)
 21     calmness = 50.0 + weight * (calmness - 50.0)
 22 
 23     # Apply the calculated happiness and calmness to the human emotion engine.
 24     human.change("happy", happiness)
 25     # Map calmness to 'friendly' as a social state
 26     human.change("friendly", calmness)
 27 
 28 
 29 def update_robot_policy(
 30     analysis: EmotionAnalysis,
 31     robot: RobotEmotionEngine,
 32 ) -> None:
 33     """
 34     Handcrafted policy that changes Albert's simulated mood.
 35 
 36     The robot does not simply copy the human's emotion. For assistive
 37     interaction, sadness or fear should generally increase Albert's
 38     friendliness and welcoming behaviour.
 39     """
 40 
 41     # If confidence is low, Albert becomes curious.
 42     if analysis.confidence < 0.30:
 43         robot.change("curious", 62)
 44         return
 45 
 46     # Adjust robot's emotions based on the detected human emotion.
 47     if analysis.emotion == "happy":
 48         robot.change("happy", 80)
 49         robot.change("welcome", 78)
 50         robot.change("friend", 72)
 51 
 52     elif analysis.emotion == "sad":
 53         robot.change("welcome", 88)
 54         robot.change("friend", 82)
 55         robot.change("curious", 62)
 56 
 57     elif analysis.emotion == "fearful":
 58         robot.change("welcome", 92)
 59         robot.change("friend", 85)
 60         robot.change("curious", 58)
 61 
 62     elif analysis.emotion == "angry":
 63         robot.change("welcome", 68)
 64         robot.change("friend", 70)
 65         robot.change("frustrated", 54)
 66 
 67     elif analysis.emotion == "frustrated":
 68         robot.change("curious", 78)
 69         robot.change("welcome", 74)
 70         robot.change("friend", 72)
 71 
 72     else:
 73         # Default behavior for other emotions or unclear states.
 74         robot.change("curious", 62)
 75         robot.change("welcome", 60)
 76

 78 def drain_battery(robot: RobotEmotionEngine) -> None:
 79     """Simulates a small battery drain after each conversation turn."""
 80     # Reduce the 'fresh' emotion, ensuring it doesn't go below 10.
 81     robot.change("fresh", max(10, robot.emotions["fresh"] - 2))

  5 def _plot_engine(
  6     axis, # The matplotlib axis object to plot on
  7     title: str, # Title for the plot
  8     engine, # The emotion engine (RobotEmotionEngine or HumanEmotionEngine)
  9     vector_label: str, # Label for the resultant emotion vector
 10 ) -> None:
 11     """Helper function to plot the emotional state of a single engine on a polar axis."""
 12     axis.set_title(title, pad=22, fontweight="bold")
 13 
 14     # Plot individual emotion axes and their current values
 15     for emotion, angle in engine.emotion_axis.items():
 16         theta = math.radians(angle) # Convert angle to radians
 17         value = engine.emotions[emotion] # Get the current emotion value
 18 
 19         axis.plot([theta, theta], [0, 100], linestyle=":", alpha=0.45) # Draw emotion axis line
 20         axis.scatter([theta], [value], s=70, zorder=3) # Plot emotion value as a scatter point
 21         axis.text(
 22             theta,
 23             111,
 24             f"{emotion}\\n{value:.0f}", # Display emotion name and value
 25             ha="center",
 26             va="center",
 27             fontsize=8,
 28         )
 29 
 30     # Calculate and plot the resultant emotion vector
 31     result_radius = float(np.clip(engine.state_r, 0, 100)) # Clip resultant radius to bounds
 32     result_theta = math.radians(engine.state_theta) # Convert resultant angle to radians
 33 
 34     axis.annotate(
 35         "",
 36         xy=(result_theta, result_radius), # End point of the arrow
 37         xytext=(0, 0), # Start point of the arrow (origin)

 39             "arrowstyle": "-|>",
 40             "linewidth": 3,

 45         [result_theta],
 46         [result_radius],
 47         marker="*",
 48         s=150,
 49         zorder=5,
 50         label=vector_label,
 51     )
 52 
 53     # Set plot limits and hide tick labels for cleanliness
 54     axis.set_ylim(0, 100)
 55     axis.set_xticklabels([])
 56     axis.set_yticklabels([])
